/* ---- Cold-start splash (components/splash.tsx) ---------------------------- */
/* Indigo screen, reversed mark: the planet appears, the rings sweep around it,
   then the two moons settle into place and the wordmark fades in. The resting
   state is the finished logo, so reduced-motion users simply see it. */
#splash {
  position: fixed;
  inset: 0;
  z-index: 9999;
  display: flex;
  flex-direction: column;
  align-items: center;
  justify-content: center;
  gap: 18px;
  background: #323490;
  transition: opacity 0.6s ease;
}
#splash[data-state="out"] {
  opacity: 0;
  pointer-events: none;
}
#splash[data-state="gone"] {
  display: none;
}
#splash .sp-mark {
  transition: transform 0.6s ease;
}
#splash[data-state="out"] .sp-mark,
#splash[data-state="out"] .sp-word {
  transform: scale(1.05);
}
#splash .sp-ring {
  stroke-dasharray: 100;
  animation: sp-sweep 1.2s cubic-bezier(0.65, 0, 0.35, 1) 0.5s both;
}
#splash .sp-ring.sp-inner {
  animation-delay: 0.7s;
}
/* the front arcs sit over the planet; they appear once the ring has swept past */
#splash .sp-front {
  animation: sp-front 1.05s linear 0s both;
}
#splash .sp-front.sp-inner {
  animation-duration: 1.25s;
}
#splash .sp-planet {
  transform-box: fill-box;
  transform-origin: center;
  animation: sp-pop 0.7s cubic-bezier(0.22, 1, 0.36, 1) 0.1s both;
}
#splash .sp-moon {
  transform-box: view-box;
  transform-origin: 50px 50px;
}
#splash .sp-moon1 {
  animation: sp-settle1 0.8s cubic-bezier(0.22, 1, 0.36, 1) 1.5s both;
}
#splash .sp-moon2 {
  animation: sp-settle2 0.8s cubic-bezier(0.22, 1, 0.36, 1) 1.7s both;
}
#splash .sp-word {
  display: flex;
  flex-direction: column;
  align-items: center;
  line-height: 1;
  transition: transform 0.6s ease;
  animation: sp-word 0.7s ease 1.5s both;
}
@keyframes sp-sweep {
  from { stroke-dashoffset: 100; }
  to { stroke-dashoffset: 0; }
}
@keyframes sp-front {
  0%, 99% { opacity: 0; }
  100% { opacity: 1; }
}
@keyframes sp-pop {
  from { opacity: 0; transform: scale(0.6); }
  to { opacity: 1; transform: scale(1); }
}
@keyframes sp-settle1 {
  from { opacity: 0; transform: rotate(-40deg); }
  to { opacity: 1; transform: rotate(0deg); }
}
@keyframes sp-settle2 {
  from { opacity: 0; transform: rotate(40deg); }
  to { opacity: 1; transform: rotate(0deg); }
}
@keyframes sp-word {
  from { opacity: 0; transform: translateY(6px); }
  to { opacity: 1; transform: translateY(0); }
}
@media (prefers-reduced-motion: reduce) {
  #splash .sp-ring,
  #splash .sp-front,
  #splash .sp-planet,
  #splash .sp-moon,
  #splash .sp-word {
    animation: none !important;
  }
}
